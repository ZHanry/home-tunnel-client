cmake_minimum_required(VERSION 3.22)
project(home_tunnel_remote VERSION 1.0.0 LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
option(HT_RD_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(HT_RD_BUILD_WORKER "Build the inherited-pipe capability worker" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(home_tunnel_remote SHARED src/remote.cpp src/protocol.cpp src/session_gate.cpp src/crypto.cpp)
target_link_libraries(home_tunnel_remote PRIVATE Threads::Threads)
target_compile_definitions(home_tunnel_remote PRIVATE HT_RD_BUILD PUBLIC HT_RD_SHARED)
target_include_directories(home_tunnel_remote PUBLIC include PRIVATE src)
set_target_properties(home_tunnel_remote PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  target_sources(home_tunnel_remote PRIVATE src/platform/windows_input.cpp)
  target_link_libraries(home_tunnel_remote PRIVATE user32 bcrypt)
endif()
if(MSVC)
  target_compile_options(home_tunnel_remote PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(home_tunnel_remote PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(HT_RD_SANITIZERS AND NOT MSVC)
  target_compile_options(home_tunnel_remote PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(home_tunnel_remote PUBLIC -fsanitize=address,undefined)
endif()

if(HT_RD_BUILD_WORKER AND NOT ANDROID)
  add_executable(home_tunnel_remote_worker src/worker.cpp)
  target_link_libraries(home_tunnel_remote_worker PRIVATE home_tunnel_remote)
endif()
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(remote_core_tests tests/core_tests.cpp src/protocol.cpp src/session_gate.cpp src/crypto.cpp)
  target_include_directories(remote_core_tests PRIVATE src include)
  target_link_libraries(remote_core_tests PRIVATE home_tunnel_remote Threads::Threads)
  if(WIN32)
    target_sources(remote_core_tests PRIVATE src/platform/windows_input.cpp)
    target_link_libraries(remote_core_tests PRIVATE user32 bcrypt)
  endif()
  if(HT_RD_SANITIZERS AND NOT MSVC)
    target_compile_options(remote_core_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(remote_core_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME remote_core_tests COMMAND remote_core_tests)
endif()
install(TARGETS home_tunnel_remote LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/home_tunnel/remote.h DESTINATION include/home_tunnel)
